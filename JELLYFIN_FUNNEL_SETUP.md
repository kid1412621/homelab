#!/bin/bash
# Installation instructions for Jellyfin Tailscale Funnel

echo "==== Jellyfin Tailscale Funnel Setup ===="
echo ""
echo "This will expose your K8s Jellyfin to Tailscale Funnel with minimal changes."
echo ""
echo "Step 1: Copy systemd service (requires sudo)"
echo "  sudo cp jellyfin-funnel.service /etc/systemd/system/"
echo ""
echo "Step 2: Enable and start the service"
echo "  sudo systemctl daemon-reload"
echo "  sudo systemctl enable jellyfin-funnel.service"
echo "  sudo systemctl restart jellyfin-funnel.service"
echo ""
echo "Step 3: Verify port-forward is working"
echo "  sleep 3 && curl http://127.0.0.1:8096"
echo ""
echo "Step 4: Configure Tailscale Funnel"
echo "  tailscale funnel --bg 127.0.0.1:8096"
echo ""
echo "Step 5: Check funnel status"
echo "  tailscale funnel status"
echo ""
echo "==== Management ===="
echo ""
echo "Check service status:"
echo "  sudo systemctl status jellyfin-funnel.service"
echo "  sudo journalctl -u jellyfin-funnel.service -f"
echo ""
echo "Disable Funnel:"
echo "  tailscale funnel reset 127.0.0.1:8096"
echo ""
echo "Disable service:"
echo "  sudo systemctl disable jellyfin-funnel.service"
echo "  sudo systemctl stop jellyfin-funnel.service"
echo ""
